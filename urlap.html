function generaltHtml(fotoUrl, adatok) { // ✅ adatok paraméter hozzáadva const htmlTartalom = `
${fotoUrl ? `` : ''}

KÉSZSÉGEK

${adatok.kesz}

ELÉRHETŐSÉG

${adatok.cim}
${adatok.tel}
${adatok.email}

${adatok.nev}

BEMUTATKOZÁS

${adatok.bemutat}

MUNKATAPASZTALAT

${adatok.tapa}

`; // Letöltés const blob = new Blob([htmlTartalom], { type: 'text/html' }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = 'Aion_Oneletrajz.html'; document.body.appendChild(a); a.click(); document.body.removeChild(a); URL.revokeObjectURL(url); }
